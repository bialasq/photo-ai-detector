   class VectorStore:
     def __init__(self, dim: int = 512):
       self.index = faiss.IndexFlatIP(dim)
       self.face_ids: list[int] = []  # mapping idx → face_id
     
     def add(self, face_id: int, emb: np.ndarray):
       norm_emb = emb / np.linalg.norm(emb)  # L2 normalize
       self.index.add(norm_emb.reshape(1, -1).astype(np.float32))
       self.face_ids.append(face_id)
     
     def search(self, emb: np.ndarray, k: int = 10) -> list[tuple[int, float]]:
       norm = emb / np.linalg.norm(emb)
       D, I = self.index.search(norm.reshape(1, -1).astype(np.float32), k)
       return [(self.face_ids[i], float(d)) for i, d in zip(I[0], D[0]) if i != -1]
     
     def save(self, path: Path):
       faiss.write_index(self.index, str(path))
       with open(path.with_suffix('.ids.pkl'), 'wb') as f:
         pickle.dump(self.face_ids, f)
     
     def load(self, path: Path):
       self.index = faiss.read_index(str(path))
       with open(path.with_suffix('.ids.pkl'), 'rb') as f:
         self.face_ids = pickle.load(f)
